cmake_minimum_required(VERSION 3.16)
project(hop_windows LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Hop Windows adapters require Windows or a Windows cross compiler")
endif()
option(HOP_WINDOWS_EXPERIMENTAL_TLS "Build the isolated Windows TLS socket adapter" OFF)
if(HOP_WINDOWS_EXPERIMENTAL_TLS)
  set(HOP_ENABLE_EXPERIMENTAL_TLS ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/../../core ${CMAKE_CURRENT_BINARY_DIR}/hop-core)
add_library(hop_windows_input STATIC src/input_sink.cpp src/input_bridge.cpp src/receiver_session_adapter.cpp src/displays.cpp src/input_ownership.cpp src/input_capture.cpp src/keymap.cpp src/clipboard.cpp src/clipboard_actor.cpp src/clipboard_worker.cpp src/clipboard_windows.cpp src/dib.cpp src/image_codec.cpp src/pointer_settings.cpp src/layout_editor.cpp src/layout_view_model.cpp src/layout_store.cpp src/live_layout_store.cpp)
target_include_directories(hop_windows_input PUBLIC include)
target_compile_features(hop_windows_input PUBLIC cxx_std_17)
target_link_libraries(hop_windows_input PUBLIC hop_core user32 windowscodecs ole32 uuid advapi32)
add_library(hop_windows_session STATIC src/session.cpp)
target_include_directories(hop_windows_session PUBLIC include)
target_compile_features(hop_windows_session PUBLIC cxx_std_17)
target_link_libraries(hop_windows_session PUBLIC hop_core)
add_library(hop_windows_host_input STATIC src/host_input_bridge.cpp src/host_capture_pump.cpp src/focus_coordinator.cpp src/live_layout.cpp src/edge_detector.cpp)
target_include_directories(hop_windows_host_input PUBLIC include)
target_compile_features(hop_windows_host_input PUBLIC cxx_std_17)
target_link_libraries(hop_windows_host_input PUBLIC hop_windows_input hop_windows_session)
add_library(hop_windows_ui_state STATIC src/ui_state.cpp)
target_include_directories(hop_windows_ui_state PUBLIC include)
target_compile_features(hop_windows_ui_state PUBLIC cxx_std_17)
add_executable(hop_windows_tray WIN32 src/tray_app.cpp)
target_link_libraries(hop_windows_tray PRIVATE hop_windows_ui_state hop_windows_input user32 shell32)
if(MINGW)
  # Bundle the GNU C++/thread runtime into the executable; Windows system DLLs
  # remain dynamically linked. This avoids requiring a separate MinGW install.
  target_link_options(hop_windows_tray PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(HOP_WINDOWS_EXPERIMENTAL_TLS)
  add_library(hop_windows_tls STATIC src/tls_socket.cpp src/session_pump.cpp src/connection.cpp src/connection_path.cpp src/connection_classifier.cpp src/lan_server.cpp src/pairing_worker.cpp src/pairing_runtime.cpp src/trust_store.cpp src/pair_transport.cpp src/pairing_review.cpp src/pairing_ui.cpp src/tcp_endpoint.cpp src/discovery.cpp)
  target_include_directories(hop_windows_tls PUBLIC include)
  target_compile_features(hop_windows_tls PUBLIC cxx_std_17)
  target_link_libraries(hop_windows_tls PUBLIC hop_tls hop_windows_session ws2_32 advapi32 user32 dnsapi)
  target_compile_definitions(hop_windows_tray PRIVATE HOP_WINDOWS_EXPERIMENTAL_TLS)
  target_link_libraries(hop_windows_tray PRIVATE hop_windows_tls hop_windows_host_input)
endif()
