cmake_minimum_required(VERSION 3.16)
project(hop_core LANGUAGES CXX)
add_library(hop_core STATIC src/layout.cpp src/focus.cpp src/input_policy.cpp src/frame.cpp src/outbound_queue.cpp src/clipboard_transfer.cpp src/clipboard_transfer_c.cpp src/wire_validation.cpp src/shortcut_translator.cpp src/shortcut_translator_c.cpp src/session.cpp src/session_c.cpp)
target_include_directories(hop_core PUBLIC include)
target_compile_features(hop_core PUBLIC cxx_std_17)
include(CTest)
option(HOP_ENABLE_EXPERIMENTAL_TLS "Build the unintegrated certificate-pinned TLS transport" OFF)
if(HOP_ENABLE_EXPERIMENTAL_TLS)
    set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
    set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    if(HOP_MBEDTLS_SOURCE_DIR)
        add_subdirectory("${HOP_MBEDTLS_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/mbedtls")
    else()
        include(FetchContent)
        FetchContent_Declare(mbedtls
            URL https://github.com/Mbed-TLS/mbedtls/releases/download/mbedtls-3.6.7/mbedtls-3.6.7.tar.bz2
            URL_HASH SHA256=a7e8bcbec0e6f761b4af24f25677626b35f762f68eef79c08677a363212d11f6)
        FetchContent_MakeAvailable(mbedtls)
    endif()
    add_library(hop_tls STATIC secure/secure_channel.cpp secure/secure_channel_c.cpp secure/tls_identity.cpp secure/pair_offer.cpp)
    if(UNIX)
        target_sources(hop_tls PRIVATE secure/posix_tls_socket.cpp secure/posix_tls_socket_c.cpp secure/posix_session_pump.cpp)
        target_link_libraries(hop_tls PUBLIC hop_core)
    endif()
    target_include_directories(hop_tls PUBLIC include)
    target_compile_features(hop_tls PUBLIC cxx_std_17)
    target_link_libraries(hop_tls PUBLIC mbedtls mbedx509 mbedcrypto)
    if(BUILD_TESTING)
        add_executable(hop_tls_tests tests/tls_tests.cpp)
        target_link_libraries(hop_tls_tests PRIVATE hop_tls)
        add_executable(hop_identity_fixture_tool tests/identity_fixture_tool.cpp)
        target_link_libraries(hop_identity_fixture_tool PRIVATE hop_tls)
        add_executable(hop_pair_offer_vector_tool tests/pair_offer_vector_tool.cpp)
        target_link_libraries(hop_pair_offer_vector_tool PRIVATE hop_tls)
        if(UNIX)
            if(HOP_ENABLE_HARMONY_FAKE_TESTS)
                add_executable(hop_harmony_receiver_network_tests ../apps/harmonyos/tests/v2_receiver_network_tests.cpp ../apps/harmonyos/entry/src/main/cpp/v2_receiver.cpp)
                target_include_directories(hop_harmony_receiver_network_tests PRIVATE ../apps/harmonyos/tests/fake ../apps/harmonyos/entry/src/main/cpp)
                target_link_libraries(hop_harmony_receiver_network_tests PRIVATE hop_tls)
                add_executable(hop_harmony_fake_tls_tests ../apps/harmonyos/tests/v2_tls_injector_tests.cpp)
                target_include_directories(hop_harmony_fake_tls_tests PRIVATE ../apps/harmonyos/tests/fake ../apps/harmonyos/entry/src/main/cpp)
                target_link_libraries(hop_harmony_fake_tls_tests PRIVATE hop_tls)
            endif()
            add_executable(hop_posix_session_pump_tests tests/posix_session_pump_tests.cpp)
            target_link_libraries(hop_posix_session_pump_tests PRIVATE hop_tls)
            add_executable(hop_posix_tls_tests tests/posix_tls_tests.cpp)
            target_link_libraries(hop_posix_tls_tests PRIVATE hop_tls)
        endif()
    endif()
endif()
if(BUILD_TESTING)
    add_executable(hop_core_tests tests/core_tests.cpp)
    target_link_libraries(hop_core_tests PRIVATE hop_core)
    add_test(NAME hop_core_tests COMMAND hop_core_tests)
    add_executable(hop_session_tests tests/session_tests.cpp)
    target_link_libraries(hop_session_tests PRIVATE hop_core)
    add_test(NAME hop_session_tests COMMAND hop_session_tests)
    add_executable(hop_session_c_tests tests/session_c_tests.cpp)
    target_link_libraries(hop_session_c_tests PRIVATE hop_core)
    add_test(NAME hop_session_c_tests COMMAND hop_session_c_tests)
    add_executable(hop_wire_vector_tool tests/wire_vector_tool.cpp)
    target_link_libraries(hop_wire_vector_tool PRIVATE hop_core)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME hop_golden_vectors COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/golden_vectors.py" "$<TARGET_FILE:hop_wire_vector_tool>")
    endif()
endif()
